 if(string.IsNullOrWhiteSpace(line))continue;var v=line.Split(',');var key=v[2];UnityEngine.Tilemaps.Tile tile;
 if(!images.TryGetValue(key,out tile)){
 var tex=new Texture2D(2,2,TextureFormat.RGBA32,false);tex.LoadImage(System.IO.File.ReadAllBytes(root+"pieces/"+key+".png"));tex.filterMode=FilterMode.Point;

 water.SetTile(new Vector3Int(int.Parse(v[0]),int.Parse(v[1]),0),tile);
}
